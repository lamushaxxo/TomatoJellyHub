local roleColors = {
    Murderer = Color3.fromRGB(220, 50, 50), Sheriff = Color3.fromRGB(80, 140, 255), Innocent = Color3.fromRGB(100, 200, 100), Hero = Color3.fromRGB(255, 215, 0)
}
local espHighlights = {}
local currentData = {}

local function updateESP()
for p, h in pairs(espHighlights) do if h then h:Destroy() end; espHighlights[p] = nil end
for _, p in ipairs(Players:GetPlayers()) do
if p ~= LP and p.Character then
local data = currentData[p.Name]
if data and data.Role and not data.Dead then
local actualRole = data.Role
if actualRole == "Innocent" then
local hasGun = p.Character:FindFirstChild("Gun") or (p.Backpack and p.Backpack:FindFirstChild("Gun"))
if hasGun then actualRole = "Hero" end
end
local color = roleColors[actualRole] or roleColors.Innocent
local h = Instance.new("Highlight")
h.FillColor = color; h.FillTransparency = 0.6
h.OutlineColor = color; h.OutlineTransparency = 0.1; h.Parent = p.Character
espHighlights[p] = h
end
end
end
end

espStatus.MouseButton1Click:Connect(function()
    espStatus.BackgroundColor3 = Theme.inactive; espStatus.Text = "Auto ESP: OFF"
    for p, h in pairs(espHighlights) do if h then h:Destroy() end end
    espHighlights = {}
    end)

local function fetchData()
local ok, res = pcall(function() return ReplicatedStorage.Remotes.Gameplay.GetCurrentPlayerData:InvokeServer() end)
if ok and type(res) == "table" then currentData = res; updateESP() end
end
task.spawn(function()
    ReplicatedStorage:WaitForChild("Remotes"):WaitForChild("Gameplay"):WaitForChild("PlayerDataChanged").OnClientEvent:Connect(fetchData)
    end)
fetchData()
task.spawn(function() while task.wait(1) do if espStatus.Text == "Auto ESP: ON" then updateESP() end end end)

-- Gun ESP & Auto-Grab Logic
task.spawn(function()
    local gunHighlight = nil
    while task.wait(0.2) do
-- Логика Gun ESP
    if gunESPOn then
    local droppedGun = findDroppedGun()
    if droppedGun then
    if not gunHighlight or gunHighlight.Adornee ~= droppedGun then
    if gunHighlight then gunHighlight:Destroy() end
    gunHighlight = Instance.new("Highlight")
    gunHighlight.FillColor = Color3.fromRGB(255, 215, 0)
    gunHighlight.OutlineColor = Color3.fromRGB(255, 255, 255)
    gunHighlight.Adornee = droppedGun; gunHighlight.Parent = droppedGun
    end
    else
        if gunHighlight then gunHighlight:Destroy(); gunHighlight = nil end
    end
    elseif gunHighlight then
    gunHighlight:Destroy(); gunHighlight = nil
    end

-- Логика Auto-Grab
    if autoGrabOn then
    local myData = currentData[LP.Name]
    local isDead = myData and myData.Dead
    local isMurderer = myData and myData.Role == "Murderer"

-- Не подбираем, если мертвы ИЛИ мы мурдер
    if not isDead and not isMurderer then
    local char = LP.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if hrp then
    local droppedGun = findDroppedGun()
    if droppedGun and droppedGun:IsDescendantOf(Workspace) then
    local dist = (hrp.Position - droppedGun.Position).Magnitude
-- Телепорт только если мы не стоим прямо на ней
    if dist <= 100 and dist > 3 then
    local originalPos = hrp.CFrame
    hrp.CFrame = CFrame.new(droppedGun.Position + Vector3.new(0, 2.5, 0))
    task.wait(0.3)
    if hrp and hrp.Parent then
    hrp.CFrame = originalPos
    end
    end
    end
    end
    end
    end
    end
    end)

local function tpTo(player)
if player == LP then return end
local myHRP = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")

if myHRP and tHRP then myHRP.CFrame = tHRP.CFrame * CFrame.new(0, 0, 3) end
end

end
end

UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if gameProcessed or not silentAimOn then return end
    if input.UserInputType ~= Enum.UserInputType.MouseButton1 and input.UserInputType ~= Enum.UserInputType.Touch then return end

    local gun = LP.Character and LP.Character:FindFirstChild("Gun")
    if not gun then return end

    local targetHead = nil
    local shortestDist = math.huge

    if LP.Character and LP.Character:FindFirstChild("HumanoidRootPart") then
    local myPos = LP.Character.HumanoidRootPart.Position
    for _, p in ipairs(Players:GetPlayers()) do
    if p ~= LP and p.Character and currentData[p.Name] and currentData[p.Name].Role == "Murderer" and not currentData[p.Name].Dead then
    local head = p.Character:FindFirstChild("Head")
    if head then
    local dist = (myPos - head.Position).Magnitude
    if dist < shortestDist then shortestDist = dist; targetHead = head end
    end
    end
    end
    end

    if not targetHead then return end

-- ПРЕДИКТ
    local ping = LP:GetNetworkPing() / 1000
    local targetVelocity = targetHead.AssemblyLinearVelocity or Vector3.new(0,0,0)
    local predictedPos = targetHead.Position + (targetVelocity * ping * predictValue)

    if gun:FindFirstChild("Shoot") then
    local hrp = LP.Character:FindFirstChild("HumanoidRootPart")
    local attach = hrp and hrp:FindFirstChild("GunRaycastAttachment")
    if attach then
    gun.Shoot:FireServer(attach.WorldCFrame, CFrame.new(predictedPos))
    end
    end
    end)

RunService.RenderStepped:Connect(function()
    if not main.Visible or not aimPage.Visible then return end
    local closestDist = math.huge
    local targetName = "None"

    for _, p in ipairs(Players:GetPlayers()) do
    if p ~= LP and p.Character and currentData[p.Name] and currentData[p.Name].Role == "Murderer" and not currentData[p.Name].Dead then
    local hrp = p.Character:FindFirstChild("HumanoidRootPart")
    if hrp and LP.Character and LP.Character:FindFirstChild("HumanoidRootPart") then
    local dist = (LP.Character.HumanoidRootPart.Position - hrp.Position).Magnitude
    if dist < closestDist then closestDist = dist; targetName = p.DisplayName end
    end
    end
    end

    if targetName ~= "None" then
    aimStatus.Text = string.format("Target: %s | Dist: %d", targetName, math.floor(closestDist))
    aimStatus.TextColor3 = Theme.primary
    else
        aimStatus.Text = "No Murderer Alive"
    aimStatus.TextColor3 = Theme.text
    end
    end)

    end)
