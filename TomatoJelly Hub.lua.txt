local function getViewport()

local function applyTheme(name)
    currentTheme = name
    local t = Themes[name]
    main.BackgroundColor3 = t.bg
    titleBar.BackgroundColor3 = t.panel
    sidebar.BackgroundColor3 = t.panel
    content.BackgroundColor3 = t.bg
    titleLabel.TextColor3 = t.text
    closeBtn.BackgroundColor3 = t.accent
    stroke.Color = t.accent
    for _, obj in ipairs(gui:GetDescendants()) do
        if obj:GetAttribute("RoleText") and obj:IsA("TextLabel") then obj.TextColor3 = t.text end
        if obj:GetAttribute("RoleAccent") and obj:IsA("GuiObject") then obj.BackgroundColor3 = t.accent end
        if obj:GetAttribute("RolePanel") and obj:IsA("GuiObject") then obj.BackgroundColor3 = t.panel end
    end
end

    knob.BackgroundColor3 = Color3.new(1,1,1)

    knob.ZIndex = 4
    knob.Parent = bar

    local function update(x)
        local rel = math.clamp((x - bar.AbsolutePosition.X) / bar.AbsoluteSize.X, 0, 1)
        local val = math.floor(min + (max - min) * rel + 0.5)
        state.value = val
        valueLabel.Text = tostring(val)
        fill.Size = UDim2.fromScale(rel, 1)
        knob.Position = UDim2.fromScale(rel, 0.5)
        if callback then callback(val) end
    end

        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true; update(input.Position.X)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            update(input.Position.X)
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then dragging = false end
    end)
    return state
end

-- ЛОГИКА
local state = { speed = 16, jump = 50, flySpeed = 50, noclip = false, fly = false, infiniteJump = false }
local noclipConn, flyConn, bodyVelocity, jumpConn

local function getHumanoid()
    local char = player.Character
    if not char then return nil end
    return char:FindFirstChildOfClass("Humanoid")
end
local function applyMovement()
    local h = getHumanoid()
    if not h then return end
    h.WalkSpeed = state.speed; h.JumpPower = state.jump; h.UseJumpPower = true
end
local function bindCharacter(char)
    local h = char:WaitForChild("Humanoid", 5)
    if not h then return end
    applyMovement()
    if jumpConn then jumpConn:Disconnect() end
    jumpConn = h.JumpRequest:Connect(function()
        if state.infiniteJump then h:ChangeState(Enum.HumanoidStateType.Jumping) end
    end)
end
player.CharacterAdded:Connect(bindCharacter)
if player.Character then bindCharacter(player.Character) end

local function setNoclip(on)
    state.noclip = on
    if noclipConn then noclipConn:Disconnect(); noclipConn = nil end
    if not on then
        local char = player.Character
        if char then
            for _, d in ipairs(char:GetDescendants()) do if d:IsA("BasePart") then d.CanCollide = true end end
            local hrp = char:FindFirstChild("HumanoidRootPart"); if hrp then hrp.CanCollide = false end
        end
        return
    end
    noclipConn = RunService.Stepped:Connect(function()
        local char = player.Character
        if not char then return end
        for _, d in ipairs(char:GetDescendants()) do if d:IsA("BasePart") then d.CanCollide = false end end
    end)
end

local flyInputs = {W=false,A=false,S=false,D=false,Space=false,Ctrl=false}
local function stopFly()
    if flyConn then flyConn:Disconnect(); flyConn = nil end
    if bodyVelocity then bodyVelocity:Destroy(); bodyVelocity = nil end
    local h = getHumanoid(); if h then h.PlatformStand = false end
end
local function startFly()
    local char = player.Character
    if not char then return end
    local hrp = char:FindFirstChild("HumanoidRootPart"); local h = getHumanoid()
    if not hrp or not h then return end
    h.PlatformStand = true
    bodyVelocity = Instance.new("BodyVelocity")
    bodyVelocity.MaxForce = Vector3.new(1e5, 1e5, 1e5); bodyVelocity.Velocity = Vector3.zero; bodyVelocity.Parent = hrp
    flyConn = RunService.Heartbeat:Connect(function()
        if not bodyVelocity or not bodyVelocity.Parent then return end
        local cam = workspace.CurrentCamera; local dir = Vector3.zero
        if flyInputs.W then dir += cam.CFrame.LookVector end
        if flyInputs.S then dir -= cam.CFrame.LookVector end
        if flyInputs.A then dir -= cam.CFrame.RightVector end
        if flyInputs.D then dir += cam.CFrame.RightVector end
        if flyInputs.Space then dir += Vector3.new(0,1,0) end
        if flyInputs.Ctrl then dir -= Vector3.new(0,1,0) end
        bodyVelocity.Velocity = dir.Magnitude > 0 and dir.Unit * state.flySpeed or Vector3.zero
    end)
end
UserInputService.InputBegan:Connect(function(input, gp)
    if gp then return end
    if input.KeyCode == Enum.KeyCode.W then flyInputs.W = true
    elseif input.KeyCode == Enum.KeyCode.A then flyInputs.A = true
    elseif input.KeyCode == Enum.KeyCode.S then flyInputs.S = true
    elseif input.KeyCode == Enum.KeyCode.D then flyInputs.D = true
    elseif input.KeyCode == Enum.KeyCode.Space then flyInputs.Space = true
    elseif input.KeyCode == Enum.KeyCode.LeftControl then flyInputs.Ctrl = true end
end)
UserInputService.InputEnded:Connect(function(input)
    if input.KeyCode == Enum.KeyCode.W then flyInputs.W = false
    elseif input.KeyCode == Enum.KeyCode.A then flyInputs.A = false
    elseif input.KeyCode == Enum.KeyCode.S then flyInputs.S = false
    elseif input.KeyCode == Enum.KeyCode.D then flyInputs.D = false
    elseif input.KeyCode == Enum.KeyCode.Space then flyInputs.Space = false
    elseif input.KeyCode == Enum.KeyCode.LeftControl then flyInputs.Ctrl = false end
end)
